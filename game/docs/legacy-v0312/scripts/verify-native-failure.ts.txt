import assert from 'node:assert/strict';
import { eq, and } from 'drizzle-orm';
import { randomUUID } from 'node:crypto';
assert.equal(new URL(process.env.DATABASE_URL!).hostname, 'feature-db');
const { db } = await import('../src/server/lib/drizzle/db');
const schema = await import('../src/server/lib/drizzle/schema');
await import('../src/server/lib/auth/auth');
const { executeDungeonCommand } = await import('../src/server/lib/services/DungeonApplicationService');
const { dungeonService } = await import('../src/server/lib/dungeon/service_v2');
const { loadCultivatorCombatInput } = await import('../src/server/lib/services/cultivator/CultivatorCombatProjectionReader');
const report = await Bun.file('output/native-realm-verification.json').json();
const actor = report.admin as { userId: string; cultivatorId: string };
const command = (command: Parameters<typeof executeDungeonCommand>[0]['command']) => executeDungeonCommand({ ...actor, command });
const started = await Promise.allSettled([command({ kind:'start',mapNodeId:'SAT_TIANLING_01' }), command({ kind:'start',mapNodeId:'SAT_TIANLING_01' })]);
assert.equal(started.filter(x=>x.status==='fulfilled').length, 1);
assert.equal((await db.select().from(schema.secretRealmRuns).where(eq(schema.secretRealmRuns.cultivatorId, actor.cultivatorId))).length, 1);
const facts = await loadCultivatorCombatInput(actor.cultivatorId);
const condition = structuredClone(facts!.cultivator.condition!);
condition.resources.hp.current = 1;
condition.resources.mp.current = 0;
await db.update(schema.cultivators).set({ condition, vitality:1,strength:1,spirit:1,endurance:1,speed:1,willpower:1 }).where(eq(schema.cultivators.id, actor.cultivatorId));
await command({kind:'action', choiceId:1, actionId:randomUUID()});
const state = await dungeonService.getState(actor.cultivatorId);
await command({kind:'battle-execute',battleId:state!.activeBattleId!,requestId:randomUUID()});
const [run] = await db.select().from(schema.dungeonRuns).where(and(eq(schema.dungeonRuns.cultivatorId,actor.cultivatorId),eq(schema.dungeonRuns.mapNodeId,'SAT_TIANLING_01')));
assert.equal(run.status,'FINISHED');
assert.equal((await db.select().from(schema.consumables).where(eq(schema.consumables.cultivatorId,actor.cultivatorId))).length,0);
console.log('PASS 两个并发入场只建立一次记录；入场后模拟濒死状态，真实战败结算不发丹');
process.exit(0);
