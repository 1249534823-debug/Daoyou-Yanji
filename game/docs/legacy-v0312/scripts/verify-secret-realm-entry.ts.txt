/** Actual isolated dungeon commands; never production accounts or state. */
import assert from 'node:assert/strict';
import { eq } from 'drizzle-orm';
assert.equal(process.env.NODE_ENV, 'test');
assert.equal(new URL(process.env.DATABASE_URL!).hostname, 'admin-test-db');
assert.equal(new URL(process.env.DATABASE_URL!).pathname, '/admin_ui_test');
assert.equal(new URL(process.env.REDIS_URL!).hostname, 'admin-test-redis');
const { db } = await import('../src/server/lib/drizzle/db');
const schema = await import('../src/server/lib/drizzle/schema');
const { EnemyGenerator } = await import('../src/shared/engine/enemyGenerator');
const { createCultivator } = await import('../src/server/lib/services/cultivator/CultivatorProfileRepository');
const { executeDungeonCommand } = await import('../src/server/lib/services/DungeonApplicationService');
const { dungeonService } = await import('../src/server/lib/dungeon/service_v2');
const { getSecretRealmMatcher } = await import('../src/server/lib/services/SecretRealmMatcher');
const { TIANLING_MAP_NODE_ID } = await import('../src/shared/config/secretRealms');
getSecretRealmMatcher(new URL('../dist/secret-realm-matcher.js', import.meta.url));
for(let i=0;getSecretRealmMatcher().health()!=='up'&&i<300;i++)await new Promise(resolve=>setTimeout(resolve,100));
assert.equal(getSecretRealmMatcher().health(),'up');
const actors: {userId:string;cultivatorId:string}[]=[];
const [previous] = await db.select().from(schema.secretRealms).where(eq(schema.secretRealms.id,'tianling'));
try {
 await db.insert(schema.secretRealms).values({id:'tianling',enabled:true}).onConflictDoUpdate({target:schema.secretRealms.id,set:{enabled:true}});
 for(let i=0;i<2;i++){
  const userId=crypto.randomUUID();
  const model=new EnemyGenerator().buildDraft({realm:'炼气',realmStage:'初期',race:'人族',difficulty:35,variantSeed:'native-test-player',name:'隔离线程验收'}).cultivator;
  const c=await createCultivator(userId,model);
  assert.ok(c.id);actors.push({userId,cultivatorId:c.id});
  await db.update(schema.cultivators).set({qi:200,qiLastRefreshedAt:new Date()}).where(eq(schema.cultivators.id,c.id));
 }
 const start=(actor:typeof actors[number])=>executeDungeonCommand({...actor,command:{kind:'start',mapNodeId:TIANLING_MAP_NODE_ID}});
 const outcomes=await Promise.allSettled([start(actors[0]),start(actors[0])]);
 assert.equal(outcomes.filter(x=>x.status==='fulfilled').length,1,JSON.stringify(outcomes.map(x=>x.status==='rejected'?String(x.reason):'ok')));
 assert.equal((await db.select().from(schema.secretRealmRuns).where(eq(schema.secretRealmRuns.cultivatorId,actors[0].cultivatorId))).length,1);
 const state=await dungeonService.getState(actors[0].cultivatorId);
 assert.equal(state?.secretRealm?.encounters.length,3);
 assert.equal(state?.secretRealm?.defeatedRounds.length,0);
 assert.equal((await db.select().from(schema.consumables).where(eq(schema.consumables.cultivatorId,actors[0].cultivatorId))).length,0);
 getSecretRealmMatcher().close();
 const [before]=await db.select().from(schema.cultivators).where(eq(schema.cultivators.id,actors[1].cultivatorId));
 await assert.rejects(start(actors[1]),/关闭/);
 const [after]=await db.select().from(schema.cultivators).where(eq(schema.cultivators.id,actors[1].cultivatorId));
 assert.equal(after.qi,before.qi);
 assert.equal((await db.select().from(schema.secretRealmRuns).where(eq(schema.secretRealmRuns.cultivatorId,actors[1].cultivatorId))).length,0);
 assert.equal(await dungeonService.getState(actors[1].cultivatorId),null);
 await Bun.write('output/secret-realm-entry-verification.json',JSON.stringify({isolated:true,concurrentEntry:true,encounters:3,noEarlyReward:true,workerFailureNoCharge:true}));
 console.log('SECRET_REALM_ENTRY_PASS');
} finally {
 getSecretRealmMatcher().close();
 if(previous)await db.update(schema.secretRealms).set({enabled:previous.enabled}).where(eq(schema.secretRealms.id,'tianling'));
 for(const actor of actors)await db.delete(schema.cultivators).where(eq(schema.cultivators.id,actor.cultivatorId));
}
process.exit(0);
