async function api(
  method: string,
  body?: unknown,
  cookie = fixture.admin.cookie,
) {
  return app.request(
    origin +
      '/api/admin/secret-realms' +
      (method === 'PATCH' ? '/tianling' : ''),
    {
      method,
      headers: {
        origin,
        'content-type': 'application/json',
        ...(cookie ? { cookie } : {}),
      },
      ...(body ? { body: JSON.stringify(body) } : {}),
    },
  );
}
async function setting() {
  const [row] = await db
    .select()
    .from(s.secretRealms)
    .where(eq(s.secretRealms.id, 'tianling'));
  return row;
}
async function patch(rewards: unknown, enabled = true) {
  const current = await setting();
  return api('PATCH', { enabled, version: current.version, rewards });
}
async function actor() {
  const userId = crypto.randomUUID();
  const model = new EnemyGenerator().buildDraft({
    realm: '炼气',
    realmStage: '初期',
    race: '人族',
    difficulty: 35,
    variantSeed: 'realm-reward-check',
    name: '隔离奖励验收',
  }).cultivator;
  const c = await createCultivator(userId, model);
  assert.ok(c.id);
  const a = { userId, cultivatorId: c.id };
  actors.push(a);
  await db
    .update(s.cultivators)
    .set({ qi: 200, qiLastRefreshedAt: new Date(), spirit_stones: 1000 })
    .where(eq(s.cultivators.id, c.id));
  return a;
}

type Prepared = {
  persist: (
    tx: DbTransaction,
  ) => Promise<
    | import('../src/server/lib/dungeon/service_v2').DungeonPersistenceSettlement
    | void
  >;
  realGains: unknown;
};
async function prepare(
  a: Actor,
  rounds: number[],
  disposition: 'completed' | 'retreated_after_battle' = 'completed',
) {
  const state = await dungeonService.getState(a.cultivatorId);
  assert.ok(state?.secretRealm);
  state.secretRealm.defeatedRounds = rounds;
  state.currentRound = 3;
  return (
    dungeonService as unknown as {
      performSettlement: (
        state: DungeonState,
        opts: unknown,
      ) => Promise<Prepared>;
    }
  ).performSettlement(state, {
    deferPersistence: true,
    endDisposition: disposition,
  });
}
async function commit(a: Actor, prepared: Prepared) {
  return withRedisLock(
    {
      key: redisLockKeys.cultivatorMutation(a.cultivatorId),
      context: 'realm_check',
      retries: 0,
    },
    (lease) =>
      playerCommandExecutor.execute({
        coordination: { mode: 'redis', lease },
        userId: a.userId,
        cultivatorId: a.cultivatorId,
        source: 'realm_isolated_acceptance',
        allowEmpty: true,
        command: (tx) =>
          executeDungeonPersistenceCommand({
            cultivatorId: a.cultivatorId,
            result: {},
            persist: prepared.persist,
            tx,
          }),
      }),
  );
}
