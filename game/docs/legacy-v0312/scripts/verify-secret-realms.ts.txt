async function fixture(email: string) {
  const id = randomUUID();
  await db.insert(authUsers).values({ id, name: '秘境验收道友', email, emailVerified: true });
  const password = randomUUID();
  await db.insert(authAccounts).values({ id: randomUUID(), userId: id, accountId: id, providerId: 'credential', updatedAt: new Date(), password: await (await auth.$context).password.hash(password) });
  const [cultivator] = await db.insert(schema.cultivators).values({ userId: id, name: '秘境验收道友', prompt: '隔离测试角色', realm: '炼气', realm_stage: '初期', vitality: 10, spirit: 10, speed: 10, willpower: 10, qi: 17 }).returning();
  const login = await auth.api.signInEmail({ body: { email, password }, asResponse: true, headers: new Headers({ origin }) });
  assert.equal(login.status, 200);
  const cookie = login.headers.getSetCookie().map((x) => x.split(';')[0]).join('; ');
  assert.ok(cookie);
  return { userId: id, cultivatorId: cultivator.id, cookie };
}

async function api(path: string, actor: typeof player | null, method = 'GET', body?: unknown) {
  return app.request(`${origin}/api${path}`, { method, headers: { origin, 'content-type': 'application/json', ...(actor ? { cookie: actor.cookie } : {}) }, ...(body ? { body: JSON.stringify(body) } : {}) });
}
