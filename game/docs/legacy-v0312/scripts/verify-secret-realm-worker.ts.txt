import assert from 'node:assert/strict';
import { SecretRealmMatcher } from '../src/server/lib/services/SecretRealmMatcher';
import { matchSecretRealmEncounters } from '../src/server/lib/services/SecretRealmCombat';
import { EnemyGenerator } from '../src/shared/engine/enemyGenerator';
assert.equal(process.env.NODE_ENV, 'test');
const url = new URL('../dist/secret-realm-matcher.js', import.meta.url);
const matcher = new SecretRealmMatcher({ workerUrl: url });
for(let i=0;matcher.health()!=='up'&&i<300;i++)await new Promise(resolve=>setTimeout(resolve,100));
assert.equal(matcher.health(),'up');
const now = new Date('2026-09-11T00:00:00Z');
const rows = [];
try {
  for (const realm of ['炼气', '金丹', '化神'] as const) {
    const player = new EnemyGenerator().buildDraft({ realm, realmStage: '中期', difficulty: 50, race: '人族', variantSeed: `worker:${realm}` }).cultivator;
    const seed = `worker-plan:${realm}`;
    const expected = matchSecretRealmEncounters(player, seed, now);
    let ticks = 0;
    let maxDelay = 0;
    let last = performance.now();
    const timer = setInterval(() => { const current = performance.now(); maxDelay = Math.max(maxDelay, current - last); last = current; ticks++; }, 20);
    const begin = performance.now();
    let actual;
    try { actual = await matcher.match(player, seed, now); } finally { clearInterval(timer); }
    assert.deepEqual(actual, expected, 'worker must preserve matching result');
    assert.ok(ticks > 5, 'main event loop must remain responsive during preview');
    rows.push({ realm, ms: performance.now() - begin, ticks, maxDelay });
  }
} finally { matcher.close(); }
await assert.rejects(matcher.match({} as never, 'closed'));
// Deliberately stalled/crashing workers exist only in this isolated test script output.
const stall = new URL('../output/stalled-matcher.js', import.meta.url);
await Bun.write(stall, "self.postMessage({type:'ready'});self.onmessage=()=>{};");
const stalled = new SecretRealmMatcher({ workerUrl: stall, queueTimeoutMs: 100, executionTimeoutMs: 300 });
const outcomes = await Promise.allSettled(Array.from({ length: 8 }, (_, i) => stalled.match({} as never, String(i))));
assert.equal(outcomes.filter(x => x.status === 'rejected').length, 8);
assert.ok(outcomes.some(x => x.status === 'rejected' && /人数/.test(x.reason.message)));
assert.ok(outcomes.some(x => x.status === 'rejected' && /排队超时/.test(x.reason.message)));
stalled.close();
const crash = new URL('../output/crashing-matcher.js', import.meta.url);
await Bun.write(crash, "self.postMessage({type:'ready'});self.onmessage=()=>{throw new Error('injected crash')};");
const crashing = new SecretRealmMatcher({ workerUrl: crash });
await assert.rejects(crashing.match({} as never, 'crash'));
crashing.close();
const shutdown = new SecretRealmMatcher({ workerUrl: stall });
const pending = shutdown.match({} as never, 'shutdown');
shutdown.close();
await assert.rejects(pending);
await Bun.write('output/secret-realm-worker-verification.json', JSON.stringify({ rows, parity: true, boundedAdmission: true, timeout: true, crash: true, shutdown: true }, null, 2));
console.log(JSON.stringify({ rows, faults: 'PASS' }));
process.exit(0);
